# arrow_config.txt - real, human-editable CAMERA config, read by
# &.widgits/board-viewer's own ops (bv_menu_input.c for the arrow
# mapping, bv_render_3d.c for the camera geometry below). No rebuild
# needed - edit a value, save, relaunch the game to pick it up.
#
# --- Arrow-key direction mapping ---
# If LEFT/RIGHT (or UP/DOWN) ever feel backwards on screen, just flip
# the matching sign below (-1 <-> 1) and relaunch. Applies to EVERY
# camera mode identically (arrows always move the real underlying grid
# position the same way - what differs mode-to-mode is how that
# position gets turned into a camera EYE position, see below).
left_dx=-1
right_dx=1
up_dy=-1
down_dy=1

# --- First-person (camera_mode 1) ---
# fp_face_distance: how far forward of the anchor cell's own center the
#   eye sits (prevents standing dead-center inside your own cell).
# fp_eye_height: how far above the ground (anchor_h) the eye sits.
fp_face_distance=0.3
fp_eye_height=1.5

# --- Third-person (camera_mode 2) ---
# tp_distance: how far BEHIND the anchor the camera sits.
# tp_height: how far ABOVE the anchor the camera sits.
# tp_look_down_deg: how many degrees to tilt the VIEW down (on top of
#   whatever r/t pitch you've dialed in) so the elevated camera
#   actually looks down at the entity instead of level over its head.
tp_distance=5.0
tp_height=4.0
tp_look_down_deg=20.0

# --- Lighting engine (xyz-ngn-plan.md) ---
# lighting_enabled: 1 = real sun/moon-driven sky color + ground light
#   level + visible sun/moon objects. 0 = flat sky, flat lighting (old
#   behavior), sun/moon objects hidden. Toggle without a rebuild.
lighting_enabled=1

# --- Camera/movement keys (all values are the KEY'S OWN ASCII CODE,
#     not a character - e.g. 'w' is 119, 'q' is 113. Use `python3 -c
#     "print(ord('x'))"` to find any key's code, or man ascii.) ---
# Defaults below match this file's own real, original hardcoded keys
# exactly - nothing changes for anyone who doesn't touch this section.
key_toggle_render_mode=48   # '0' - flip between 2D and 3D
key_possess=57              # '9' - possess/release the hero
key_reset_xelector=56       # '8' - snap xelector back to the hero
key_z_down=122              # 'z' - xelector Z-level down
key_z_up=120                # 'x' - xelector Z-level up
key_reset_view=102          # 'f' - reset camera to this mode's default
key_yaw_left=113            # 'q' - turn left (modes 1/2/3)
key_yaw_right=101           # 'e' - turn right (modes 1/2/3)
key_pitch_down=114          # 'r' - look down (modes 1/2/3)
key_pitch_up=116            # 't' - look up (modes 1/2/3)
key_pan_forward=119         # 'w' - pan/strafe forward (modes 3/4)
key_pan_back=115            # 's' - pan/strafe back (modes 3/4)
key_pan_left=97             # 'a' - pan/strafe left (modes 3/4)
key_pan_right=100           # 'd' - pan/strafe right (modes 3/4)
key_cam_down=99             # 'c' - free camera height down (modes 3/4)
key_cam_up=118              # 'v' - free camera height up (modes 3/4)
# NOTE: the '1'/'2'/'3'/'4' camera-mode-switch keys are NOT yet
# remappable here - they use a compact "key - '0' = mode number" trick
# in bv_menu_input.c that doesn't generalize to arbitrary custom keys
# without a bigger real refactor. Flagged honestly, not silently
# skipped - ask if you actually need those remapped and it's a real,
